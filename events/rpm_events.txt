namespace = rpm_events

###############################################################################
# rpm_events.1 - the post-war summary
#
# Fired once the player holds snapshotted states again.
#
# All bookkeeping happens in immediate, not in the on_action that fires this.
# Paradox documents that an on_action's effect block and the events it fires are
# separate chains, so a variable written there is not reliably readable here.
#
# Option keys are named .opt_* rather than .a/.b/.c. An option called
# rpm_events.1.d collides with the event's own desc key, and the game silently
# renders the whole description inside that button instead of its label.
#
# Every option's work sits inside hidden_effect. Victoria 3 builds an option's
# tooltip by evaluating its effects without committing the writes, so an effect
# chain that reads a variable it set in the same pass floods the error log with
# "Failed to fetch variable ... due to not being set". hidden_effect keeps the
# tooltip pass out of it entirely.
###############################################################################
rpm_events.1 = {
	type = country_event
	placement = ROOT

	title = rpm_events.1.t
	desc = rpm_events.1.d
	flavor = rpm_events.1.f

	event_image = {
		video = "unspecific_gears_pistons"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	duration = 3

	immediate = {
		# Only on the first showing - reopening must not reset the counter, or
		# the cap that stops an endless popup would never be reached.
		if = {
			limit = { NOT = { has_variable = rpm_summary_shows } }
			set_variable = { name = rpm_summary_shows value = 1 }
		}
		rpm_build_summary = yes
	}

	option = {
		name = rpm_events.1.opt_pms
		hidden_effect = {
			rpm_restore_pms = yes
			rpm_reopen_summary_or_finish = yes
		}
	}

	option = {
		name = rpm_events.1.opt_levels
		hidden_effect = {
			rpm_restore_levels_general = yes
			rpm_reopen_summary_or_finish = yes
		}
	}

	option = {
		name = rpm_events.1.opt_army
		hidden_effect = {
			rpm_restore_levels_army = yes
			rpm_reopen_summary_or_finish = yes
		}
	}

	option = {
		name = rpm_events.1.opt_conscription
		hidden_effect = {
			rpm_restore_levels_conscription = yes
			rpm_reopen_summary_or_finish = yes
		}
	}

	option = {
		name = rpm_events.1.opt_done
		default_option = yes
		hidden_effect = {
			rpm_mark_states_reported = yes
		}
	}
}

###############################################################################
# rpm_events.9 - diagnostics
#
# Nothing fires this, which the game notes as "Event rpm_events.9 is orphaned".
# That is intended: open the console in debug mode and run
#     event rpm_events.9
# to see whether snapshots exist and what the mod currently counts, without
# having to fight a whole civil war first.
#
# It also toggles verbose logging. With it on, every difference the mod finds
# and every change it makes is written to the game log naming the state and the
# building type - which is the only practical way to see which building in which
# state was actually touched. Turn it on, then run this event again: the recount
# in immediate dumps the whole picture.
###############################################################################
rpm_events.9 = {
	type = country_event
	placement = ROOT

	title = rpm_events.9.t
	desc = rpm_events.9.d

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	duration = 3

	immediate = {
		rpm_debug_count_states = yes
	}

	option = {
		name = rpm_events.9.opt_close
		default_option = yes
	}

	# Verbose logging writes one line per difference found and per change made,
	# naming the state and the building type, into the game log. It costs
	# nothing while it is off, and debug_log only writes at all when the game is
	# running in debug mode.
	option = {
		name = rpm_events.9.opt_log_on
		hidden_effect = {
			set_global_variable = rpm_debug
			debug_log = "RPM verbose logging enabled"
		}
	}

	option = {
		name = rpm_events.9.opt_log_off
		hidden_effect = {
			debug_log = "RPM verbose logging disabled"
			remove_global_variable = rpm_debug
		}
	}
}
