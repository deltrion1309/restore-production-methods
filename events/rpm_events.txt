namespace = rpm_events

###############################################################################
# rpm_events.1 - the post-war summary
#
# Fired once the player holds snapshotted states again.
#
# Every option's work sits inside hidden_effect. Victoria 3 builds an option's
# tooltip by evaluating its effects without committing the writes, so an effect
# chain that reads a variable it set in the same pass floods the error log with
# "Failed to fetch variable ... due to not being set". hidden_effect keeps the
# tooltip pass out of it entirely.
###############################################################################
rpm_events.1 = {
	type = country_event
	placement = ROOT

	title = rpm_events.1.t
	desc = rpm_events.1.d
	flavor = rpm_events.1.f

	event_image = {
		video = "unspecific_gears_pistons"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	duration = 3

	option = {
		name = rpm_events.1.a
		hidden_effect = {
			rpm_restore_pms = yes
			rpm_build_summary = yes
			rpm_reopen_summary_or_finish = yes
		}
	}

	option = {
		name = rpm_events.1.b
		default_option = yes
		hidden_effect = {
			rpm_mark_states_reported = yes
		}
	}
}

###############################################################################
# rpm_events.9 - diagnostics
#
# Not fired by anything. Open the console in debug mode and run
#     event rpm_events.9
# to see whether snapshots exist and what the mod currently counts, without
# having to fight a whole civil war first.
###############################################################################
rpm_events.9 = {
	type = country_event
	placement = ROOT

	title = rpm_events.9.t
	desc = rpm_events.9.d

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	duration = 3

	immediate = {
		rpm_debug_count_states = yes
	}

	option = {
		name = rpm_events.9.a
		default_option = yes
	}

	# Runs both candidate spellings of add_building_level with a delta of 0, so
	# nothing changes and error.log names whichever one is wrong.
	option = {
		name = rpm_events.9.b
		hidden_effect = {
			capital ?= {
				rpm_probe_add_building_level_type = yes
				rpm_probe_add_building_level_building = yes
			}
		}
	}
}
