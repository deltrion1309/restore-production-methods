###############################################################################
# Restore Production Methods - shared conditions
#
# Every var: read is guarded by has_variable. Reading an unset variable is an
# error in Victoria 3, not a silent zero.
###############################################################################

# Scope: state
# A state that was snapshotted when it defected and has not been reported yet.
rpm_state_is_snapshotted = {
	has_variable = rpm_pending
}

# Scope: state
# A snapshotted state that is back in the hands of the country it defected from.
# Comparing against the state's own owner keeps this self-contained, and means a
# third country conquering the state does not trip the summary.
rpm_state_is_returned = {
	has_variable = rpm_pending
	var:rpm_original_owner ?= owner
}

# Scope: country
rpm_has_returned_states = {
	any_scope_state = {
		rpm_state_is_returned = yes
	}
}

# Scope: country
# Is there anything left for the player to put back? This is what stops the
# summary reopening forever once every restore has been applied.
rpm_has_anything_to_restore = {
	OR = {
		AND = {
			has_variable = rpm_sum_pm_changes
			var:rpm_sum_pm_changes > 0
		}
		AND = {
			has_variable = rpm_sum_buildings_expanded
			var:rpm_sum_buildings_expanded > 0
		}
	}
}
