###############################################################################
# Restore Production Methods - Phase 3, the restore actions
#
# add_building_level does not exist in 1.13 - the game logs "Unknown effect
# add_building_level" even though it ships a tooltip key for it. The only way to
# resize a building is vanilla's own idiom, remove_building followed by
# create_building at the wanted level, and that resets the building's production
# methods.
#
# So every level restore reapplies the production method snapshot afterwards.
# Levels first, methods second - never the other way round.
###############################################################################

# Scope: country. Put production methods back to the snapshot in every state
# that has come back. The per-building-type work is generated.
rpm_restore_pms = {
	every_scope_state = {
		limit = { rpm_state_is_returned = yes }
		rpm_generated_restore_pms_state = yes
	}
}

# Scope: country. Put the army back to the size it was before the revolt.
#
# Barracks are not sized by building them: you set an army size in the army view
# and the game builds or downsizes Barracks to match. Shrinking the buildings
# while the target stayed high just invited the game to rebuild them, which is
# why the old barracks button never appeared to do anything.
#
# The target is country-wide rather than per-state. There is no way around that:
# a military formation exposes its front, country, commander and headquarters to
# script, but not its size, so per-army targets cannot be read or written.
rpm_restore_army_size = {
	if = {
		limit = { has_variable = rpm_snap_army_size }
		set_country_army_sp_target = var:rpm_snap_army_size
		if = {
			limit = { has_global_variable = rpm_debug }
			debug_log = "RPM army target set back to [THIS.GetCountry.MakeScope.Var('rpm_snap_army_size').GetValue|0]"
		}
	}
}

# Scope: country. The same for the navy.
rpm_restore_navy_size = {
	if = {
		limit = { has_variable = rpm_snap_navy_size }
		set_country_navy_sp_target = var:rpm_snap_navy_size
		if = {
			limit = { has_global_variable = rpm_debug }
			debug_log = "RPM navy target set back to [THIS.GetCountry.MakeScope.Var('rpm_snap_navy_size').GetValue|0]"
		}
	}
}

# Scope: country. Shrink every expandable building back to its pre-war level,
# undoing the rebel AI's expansions. Buildings the AI shrank are left alone,
# buildings it built from scratch are never demolished, and Barracks and
# Conscription Centres are not touched here - they follow the army size target.
rpm_restore_levels_general = {
	every_scope_state = {
		limit = { rpm_state_is_returned = yes }
		rpm_generated_restore_levels_general_state = yes
	}
	rpm_restore_pms = yes
}

# Scope: country.
# A Victoria 3 event closes as soon as an option is picked, so applying one
# restore and then reopening the summary is how several restores are offered in
# turn. Two things stop that becoming a loop:
#
#   1. it only reopens while something is actually left to restore, and
#   2. a hard cap on how many times the summary may reopen, so a restore that
#      silently fails to change anything cannot trap the player.
rpm_reopen_summary_or_finish = {
	change_variable = { name = rpm_summary_shows add = 1 }

	if = {
		limit = {
			var:rpm_summary_shows < rpm_max_summary_shows
			rpm_has_anything_to_restore = yes
		}
		trigger_event = { id = rpm_events.1 popup = yes }
	}
	else = {
		rpm_mark_states_reported = yes
	}
}

###############################################################################
# Diagnostics - only reachable from the debug event (console: event rpm_events.9)
###############################################################################

# Scope: country
rpm_debug_count_states = {
	save_scope_as = rpm_player

	set_variable = { name = rpm_dbg_pending value = 0 }
	set_variable = { name = rpm_dbg_reported value = 0 }

	every_scope_state = {
		if = {
			limit = { has_variable = rpm_pending }
			scope:rpm_player = { change_variable = { name = rpm_dbg_pending add = 1 } }
		}
		if = {
			limit = { has_variable = rpm_reported }
			scope:rpm_player = { change_variable = { name = rpm_dbg_reported add = 1 } }
		}
	}

	rpm_build_summary = yes
}
