###############################################################################
# Restore Production Methods - Phase 2, the post-war summary
#
# Counts what changed between the snapshot and now, across every state that has
# come back. The counters live as country variables so the event text can print
# them. The per-building comparison is in rpm_generated_effects.txt.
###############################################################################

# Scope: country
rpm_reset_summary = {
	set_variable = { name = rpm_sum_states value = 0 }
	set_variable = { name = rpm_sum_buildings_expanded value = 0 }
	set_variable = { name = rpm_sum_buildings_reduced value = 0 }
	set_variable = { name = rpm_sum_buildings_new value = 0 }
	set_variable = { name = rpm_sum_levels_added value = 0 }
	set_variable = { name = rpm_sum_pm_changes value = 0 }
	set_variable = { name = rpm_sum_pm_buildings value = 0 }
	set_variable = { name = rpm_sum_pm_blocked value = 0 }
	set_variable = { name = rpm_sum_buildings_grew value = 0 }
	set_variable = { name = rpm_sum_army_delta value = 0 }
	set_variable = { name = rpm_sum_navy_delta value = 0 }
}

# Scope: country
rpm_build_summary = {
	save_scope_as = rpm_player
	rpm_reset_summary = yes

	# How much bigger the army and navy are than when the revolt broke out.
	# Retaking the rebels' states hands you their barracks, so this is usually
	# the difference the player wants undone.
	if = {
		limit = {
			has_variable = rpm_snap_army_size
			army_size > var:rpm_snap_army_size
		}
		save_temporary_scope_value_as = {
			name = rpm_army_delta
			value = {
				value = army_size
				subtract = var:rpm_snap_army_size
			}
		}
		set_variable = { name = rpm_sum_army_delta value = scope:rpm_army_delta }
	}
	if = {
		limit = {
			has_variable = rpm_snap_navy_size
			navy_size > var:rpm_snap_navy_size
		}
		save_temporary_scope_value_as = {
			name = rpm_navy_delta
			value = {
				value = navy_size
				subtract = var:rpm_snap_navy_size
			}
		}
		set_variable = { name = rpm_sum_navy_delta value = scope:rpm_navy_delta }
	}

	every_scope_state = {
		limit = { rpm_state_is_returned = yes }

		scope:rpm_player = {
			change_variable = { name = rpm_sum_states add = 1 }
		}

		# Verbose logging, off unless the diagnostics event turned it on. The
		# generated diff writes one line per difference it finds, naming the
		# state, the building type, the production method group and the method
		# it expected. debug_log only writes when the game runs in debug mode.
		if = {
			limit = { has_global_variable = rpm_debug }
			debug_log = "RPM state | [THIS.GetState.GetNameNoFormatting] | recovered with a snapshot"
		}

		rpm_generated_diff_state = yes
	}
}

# Scope: country
# Mark the returned states as reported so the monthly check stops firing for
# them. The snapshot itself is deliberately kept: Phase 3 needs it.
rpm_mark_states_reported = {
	every_scope_state = {
		limit = { rpm_state_is_returned = yes }
		remove_variable = rpm_pending
		set_variable = rpm_reported
	}
	remove_variable = rpm_summary_open
	remove_variable = rpm_summary_shows
	remove_variable = rpm_snap_army_size
	remove_variable = rpm_snap_navy_size
}
