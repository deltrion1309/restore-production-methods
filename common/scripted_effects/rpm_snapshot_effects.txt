###############################################################################
# Restore Production Methods - Phase 1, the pre-war snapshot
#
# Taken the moment a revolution or secession breaks out. By the time
# on_revolution_start / on_secession_start run, the defecting states already
# belong to the uprising country (vanilla's own Paris Commune branch inside
# on_revolution_start manipulates that country's territory), so iterating
# scope:target's states is an exact, generic way to find them - no guessing
# which states will defect and no hardcoded lists.
#
# The per-building half of the snapshot lives in rpm_generated_effects.txt,
# because building scopes cannot store variables and a building's production
# methods cannot be iterated in script. See docs/ARCHITECTURE.md.
###############################################################################

# Scope: country - the loyalist side. scope:target = the uprising country.
rpm_take_snapshot = {
	save_scope_as = rpm_loyalist

	scope:target ?= {
		every_scope_state = {
			rpm_snapshot_state = yes
		}
	}
}

# Scope: state - one defecting state. scope:rpm_loyalist must be set.
rpm_snapshot_state = {
	set_variable = { name = rpm_original_owner value = scope:rpm_loyalist }
	set_variable = rpm_pending

	rpm_generated_snapshot_state = yes
}

# Scope: state - forget a state entirely.
rpm_clear_state_snapshot = {
	remove_variable = rpm_pending
	remove_variable = rpm_reported
	remove_variable = rpm_original_owner

	rpm_generated_clear_state = yes
}
