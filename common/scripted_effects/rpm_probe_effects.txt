# changes nothing. Whichever one is wrong shows up in error.log, and then the
# level restores can be generated with the right one.
#
# Scope for both: state.
###############################################################################

	add_building_level = {
		type = building_barrack
		level = 0
	}
}

rpm_probe_add_building_level_building = {
	add_building_level = {
		building = building_barrack
		level = 0
	}
}
